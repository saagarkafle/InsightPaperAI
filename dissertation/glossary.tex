% ==========================================
% Glossary Terms (\printglossary)
% ==========================================
\newglossaryentry{vectorembedding}{
    name={Vector Embedding},
    description={A continuous high-dimensional numerical representation of text where geometric distance corresponds to semantic similarity.}
}

\newglossaryentry{cosinesimilarity}{
    name={Cosine Similarity},
    description={A mathematical metric measuring the normalized inner product between two vectors to quantify contextual similarity.}
}

\newglossaryentry{hallucination}{
    name={Hallucination},
    description={A phenomenon in large language models where generated text contains plausible-sounding but factually incorrect or ungrounded claims.}
}

\newglossaryentry{chunking}{
    name={Chunking},
    description={The process of segmenting long documents into smaller, overlapping textual passages optimized for vector embedding and retrieval.}
}

\newglossaryentry{promptengineering}{
    name={Prompt Engineering},
    description={The strategic design of natural language instructions to steer language model behavior toward structured, faithful outputs.}
}

\newglossaryentry{semanticsearch}{
    name={Semantic Search},
    description={An information retrieval technique that searches by contextual meaning rather than literal keyword matching.}
}

\newglossaryentry{contrastivelearning}{
    name={Contrastive Learning},
    description={A machine learning paradigm that trains embedding models by pulling relevant query-passage pairs together while pushing irrelevant pairs apart.}
}
